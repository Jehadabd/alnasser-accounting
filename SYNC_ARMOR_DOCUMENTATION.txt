 📦 القسم الأول: نظام رقم الفاتورة المركّب
══════════════════════════════════════════════════════════════════════════════

┌──────────────────────────────────────────────────────────────────────────┐
│  الفكرة الأساسية                                                         │
├──────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│  الفاتورة لها نوعان من المعرّفات:                                         │
│                                                                          │
│  1. الـ id الداخلي (Surrogate Key):                                     │
│     - INTEGER AUTOINCREMENT (1, 2, 3, ...)                              │
│     - للتتبع التقني الداخلي والربط بين الجداول                           │
│     - لا يُعرض للمستخدم أبداً                                            │
│                                                                          │
│  2. رقم الفاتورة التجاري (Natural Key):                                 │
│     - نص مركّب: [رقم الجهاز][السنة][الشهر][التسلسل الشهري]               │
│     - مثال: 120260815                                                   │
│       • 1 = رقم الجهاز                                                  │
│       • 2026 = السنة                                                    │
│       • 8 = الشهر (بدون صفر بادئ)                                       │
│       • 15 = التسلسل الشهري (15th فاتورة في هذا الشهر لهذا الجهاز)       │
│     - هذا هو الرقم الذي يُعرض ويُطبع ويُبحث به                          │
│                                                                          │
└──────────────────────────────────────────────────────────────────────────┘

 🔒 القسم الثاني: المزامنة المُدرّعة الذرية (Armored Atomic Sync)
══════════════════════════════════════════════════════════════════════════════

┌──────────────────────────────────────────────────────────────────────────┐
│  الثغرة المحاسبية الحرجة التي اكتُشفت وإصلاحها                            │
├──────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│  ❌ المشكلة الخطيرة:                                                     │
│                                                                          │
│  عند إنشاء فاتورة دين، كانت تُنشأ معاملة مالية (DebtTransaction) لكن:    │
│  - لم تكن تحصل على transaction_uuid (يبقى NULL)                         │
│  - لم تكن تحصل على sync_uuid (يبقى NULL)                                │
│  - لم تكن مرتبطة بالفاتورة عبر UUID                                     │
│                                                                          │
│  النتيجة الكارثية:                                                       │
│  - النظام يرفع المعاملات بشرط transaction_uuid IS NOT NULL              │
│  - معاملة الفاتورة كانت تُتجاهل!                                          │
│  - يعني: جهاز A يبيع بالدين → الفاتورة تصل لجهاز B                       │
│    لكن دين العميل لا يتأثر على جهاز B!                                   │
│  - رصيد العميل مختلف بين الجهازين ← اختلاف محاسبي حقيقي                  │
│                                                                          │
│  ✅ الحل: المزامنة الذرية للحزمة المدمجة                                 │
│                                                                          │
└──────────────────────────────────────────────────────────────────────────┘

┌──────────────────────────────────────────────────────────────────────────┐
│  مفهوم "المزامنة الذرية" (Atomic Sync)                                   │
├──────────────────────────────────────────────────────────────────────────┤
│                                                                          │

│                                                                          │
│  تُرفع معاً في وثيقة Firestore واحدة، وتُستقبل معاً في SQLite            │
│  transaction واحدة.                                                      │
│                                                                          │
│  النتيجة:                                                                │
│  إما تصل الحزمة كاملة (فاتورة + أثرها المالي) أو لا تصل أبداً.           │
│  لا حالة وسطية (فاتورة بدون معاملة، أو معاملة بدون فاتورة).              │

حقل جديد مضاف للنموذج DebtTransaction:

    final String? invoiceSyncUuid; // 🔒 ربط المعاملة بالفاتورة عبر UUID

تم إضافته في 5 مواضع:
  1. تعريف الحقل (بعد audioNotePath)
  2. Constructor: this.invoiceSyncUuid,
  3. toMap(): 'invoice_sync_uuid': invoiceSyncUuid,
  4. fromMap(): invoiceSyncUuid: map['invoice_sync_uuid'] as String?,
  5. copyWith(): this.invoiceSyncUuid, invoiceSyncUuid: invoiceSyncUuid ?? this.invoiceSyncUuid,

الغرض:

  عند رفع الفاتورة، نبحث عن معاملاتها عبر: WHERE invoice_sync_uuid = ?
  عند التحديث، نحذف المعاملات القديمة: WHERE invoice_sync_uuid = ?

    static String newTransactionUuid() {

    /// 🚀 رفع فوري لحزمة فاتورة واحدة بعد حفظها مباشرة.
    Future<bool> syncInvoiceBundleNow(String invoiceUuid) async {
      if (!await FirebaseSyncConfig.isEnabled()) return false;
      if (invoiceUuid.isEmpty) return false;

      try {
        final fullInvoice = await _coordinator.getFullInvoiceByUuid(invoiceUuid);
        if (fullInvoice == null) return false;
        if ((fullInvoice['is_synced'] as int?) == 1) return true; // مرفوعة بالفعل

        final collection = _firestore.collection('invoices');
        final payload = await _buildInvoiceBundlePayload(fullInvoice, collection);
        if (payload == null) return false;

        await collection.doc(invoiceUuid).set(payload, SetOptions(merge: true));
        await _coordinator.markAsSynced(invoiceUuid);
        await _markInvoiceTransactionsAsSynced(fullInvoice);

        print('⚡ رفع فوري ناجح لحزمة الفاتورة $invoiceUuid');
        return true;
      } catch (e) {
        print('❌ فشل الرفع الفوري للفاتورة $invoiceUuid: $e');
        return false;
      }
    }

    /// يبني حمولة (payload) وثيقة الفاتورة الكاملة المدمجة.
    Future<Map<String, dynamic>?> _buildInvoiceBundlePayload(
      Map<String, dynamic> invMap,
      CollectionReference collection,
    ) async {
      final uuid = invMap['invoice_uuid'] as String?;
      if (uuid == null || uuid.isEmpty) return null;

      final payload = Map<String, dynamic>.from(invMap);
      payload['invoice_number'] = invMap['invoice_number'];
      payload.remove('id');
      payload.remove('is_synced');

      // ربط الفاتورة بالعميل عبر sync_uuid
      payload['customer_sync_uuid'] =
          await _customerSyncUuidFor(invMap['customer_id'] as int?);
      payload.remove('customer_id');

      // items مدمجة
      final items = (invMap['items'] as List?) ?? const [];
      payload['items'] = items.map((it) {
        final m = Map<String, dynamic>.from(it as Map);
        m.remove('id');
        m.remove('invoice_id');
        return m;
      }).toList();

      // 🔒 معاملات الفاتورة المالية مدمجة (مزامنة ذرية)
      final transactions = (invMap['transactions'] as List?) ?? const [];
      payload['transactions'] = transactions.map((tx) {
        final txMap = Map<String, dynamic>.from(tx as Map);
        txMap.remove('id');
        txMap.remove('invoice_id');
        txMap.remove('customer_id');
        txMap.remove('is_uploaded');
        txMap.remove('is_read_by_others');
        return txMap;
      }).toList();

      payload['_uploaded_at'] = DateTime.now().toIso8601String();
      payload['uploadedAt'] = FieldValue.serverTimestamp();

      return payload;
    }

    /// 🔁 إعادة حساب رصيد العميل من مجموع معاملاته (شباك أمان محاسبي).
    Future<void> _recalculateCustomerBalanceInsideTxn(
        sqflite.Transaction txn, int customerId) async {
      try {
        final result = await txn.rawQuery('''
          SELECT COALESCE(SUM(amount_changed), 0) as total
          FROM transactions
          WHERE customer_id = ? AND (is_deleted IS NULL OR is_deleted = 0)
        ''', [customerId]);
        final total = (result.first['total'] as num?)?.toDouble() ?? 0.0;
        await txn.update('customers',
            {'current_total_debt': total,
             'last_modified_at': DateTime.now().toIso8601String()},
            where: 'id = ?', whereArgs: [customerId]);
      } catch (e) {
        print('⚠️ تعذّر إعادة حساب رصيد العميل $customerId: $e');
      }
    }

 🔄 القسم الرابع: كيفية عمل النظام بعد التحديث (تدفق البيانات)
══════════════════════════════════════════════════════════════════════════════

┌──────────────────────────────────────────────────────────────────────────┐
│  السيناريو 1: إنشاء فاتورة دين جديدة                                    │
├──────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│  1. المحاسب يضغط "حفظ" في POS                                           │
│  2. pos_provider.processCheckout() → DatabaseService.createPosInvoice() │
│  3. InvoiceManager.saveCompleteInvoice():                                │
│     ├── حساب التسلسل الشهري (creator_device_id + year + month)          │
│     ├── بناء invoice_number = "120260815"                                │
│     ├── توليد invoice_uuid (inv_...)                                     │
│     ├── SQLite transaction:                                              │
│     │   ├── INSERT invoice (with invoice_number, uuid, year, month)     │
│     │   ├── INSERT items                                                 │
│     │   └── INSERT transaction (with uuid, sync_uuid, invoice_sync_uuid)│
│     └── WAL checkpoint (ضمان الكتابة للقرص)                              │
│  4. العودة لـ processCheckout():                                         │
│     ├── جلب الفاتورة المحفوظة                                           │
│     ├── تعيين _lastInvoiceDisplayNumber                                 │
│     ├── ⚡ رفع فوري: InvoiceSyncService.syncInvoiceBundleNow(uuid)       │
│     │   ├── getFullInvoiceByUuid() → فاتورة + items + transactions      │
│     │   ├── _buildInvoiceBundlePayload() → وثيقة مدمجة                   │
│     │   ├── Firestore: collection.doc(uuid).set(payload)                │
│     │   ├── markAsSynced(uuid)                                          │
│     │   └── markTransactionAsSynced() لكل معاملة                         │
│     └── طباعة الإيصال الحراري (بالرقم التجاري)                           │
│                                                                          │
│  النتيجة:                                                                │
│  - المحاسب يرى الشاشة فوراً (لا انتظار للشبكة)                           │
│  - الفاتورة مرفوعة لـ Firebase خلال ثوانٍ                               │
│  - الأجهزة الأخرى تستقبلها تلقائياً                                      │
│                                                                          │
└──────────────────────────────────────────────────────────────────────────┘

┌──────────────────────────────────────────────────────────────────────────┐
│  السيناريو 2: استقبال فاتورة من جهاز آخر                                │
├──────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│  1. snapshot listener يطلق (أي تغيير في مجموعة invoices)                │
│  2. لكل وثيقة واردة → _processIncomingInvoice(uuid, data):              │
│     ├── فحص الملكية (creator_device_id == myDeviceId → skip)            │
│     ├── فحص الإصدار (localVersion >= incoming → skip)                   │
│     ├── فلترة الحقول (_invoiceColumns)                                   │
│     ├── فحص العميل (customer_sync_uuid)                                 │
│     │   └── لو غير موجود → حفظ في sync_invoice_orphans                  │
│     └── SQLite transaction:                                              │
│         ├── فحص إدمبوتنت داخل المعاملة                                   │
│         ├── INSERT/UPDATE invoice                                        │
│         ├── حذف items القديمة + INSERT items الجديدة                     │
│         ├── 🔒 حذف transactions القديمة (WHERE invoice_sync_uuid)        │
│         ├── 🔒 INSERT transactions المدمجة                               │
│         └── 🔁 إعادة حساب رصيد العميل من SUM                             │
│  3. واجهة المستخدم تتحدث تلقائياً                                        │
│                                                                          │
│  النتيجة:                                                                │
│  - الفاتورة تصل كاملة (فاتورة + items + أثرها المالي)                    │
│  - رصيد العميل محدّث ومتسق                                               │
│  - لا تكرار (الإدمبوتنت يحمي)                                            │
│                                                                          │
└──────────────────────────────────────────────────────────────────────────┘

┌──────────────────────────────────────────────────────────────────────────┐
│  السيناريو 3: فشل الرفع الفوري (شبكة ضعيفة)                             │
├──────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│  1. syncInvoiceBundleNow() يفشل                                          │
│  2. is_synced تبقى = 0                                                   │
│  3. المؤقت الدوري (كل 3 دقائق) يلتقطها:                                  │
│     syncPendingInvoices() → نفس منطق الحزمة المدمجة                       │
│  4. لو فشلت أيضاً، تُعاد في الدورة التالية                              │
│  5. لا فاتورة تضيع أبداً                                                  │
│                                                                          │
└──────────────────────────────────────────────────────────────────────────┘

┌──────────────────────────────────────────────────────────────────────────┐
│  السيناريو 4: إقلاع جهاز بعد انطفاء طويل                                │
├──────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│  1. snapshot listener يطلق بكل الفواتير المتراكمة                         │
│  2. كل فاتورة تُعالج عبر _processIncomingInvoice()                       │
│  3. الإدمبوتنت يرفض ما هو موجود محلياً بالفعل                            │
│  4. الفواتير الجديدة تُطبّق ذرياً                                         │
│  5. المعاملات اليتيمة تنتظر في sync_invoice_orphans حتى يصل العملاء      │
│                                                                          │
└──────────────────────────────────────────────────────────────────────────┘

 ⚠️ القسم الخامس: التحذيرات والتنبيهات المهمة
══════════════════════════════════════════════════════════════════════════════

┌──────────────────────────────────────────────────────────────────────────┐
│  ⚠️ تنبيه 1: الهجرة ستُعيد توليد أرقام الفواتير القديمة                  │
├──────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│  عند أول تشغيل بعد التحديث، ستُعيد الهجرة (_migrateInvoiceNumbers)       │
│  توليد أرقام كل الفواتير القديمة التي تنقصها البيانات الصحيحة.            │
│                                                                          │
│  هذا مقصود - الفواتير الحالية أرقامها خاطئة/فوضوية.                       │
│  الفواتير القديمة ستحصل على تسلسل صحيح (1, 2, 3...) حسب ترتيبها الزمني   │
│  داخل كل شهر.                                                            │
│                                                                          │
│  قد يربك المستخدم إن كان قد حفظ أرقاماً قديمة ورقياً.                     │
│                                                                          │
└──────────────────────────────────────────────────────────────────────────┘

┌──────────────────────────────────────────────────────────────────────────┐
│  ⚠️ تنبيه 2: القيد الفريد المركّب                                        │
├──────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│  أُضيف قيد فريد على (creator_device_id, invoice_year,                    │
│                       invoice_month, monthly_sequence_number)             │
│                                                                          │
│  هذا يمنع تكرار نفس الرقم لنفس الجهاز في نفس الشهر.                       │
│  لو حصل تعارض (نادر جداً)، الهجرة تُصلحه لأنها تعيد التسلسل من الصفر.     │
│                                                                          │
│  لا تعارض ممكن بين الأجهزة لأن رقم الجهاز جزء من المفتاح.                │
│  (جهاز 1 ينتج 1202681، جهاز 2 ينتج 2202681 - مختلفان)                    │
│                                                                          │
└──────────────────────────────────────────────────────────────────────────┘

┌──────────────────────────────────────────────────────────────────────────┐
│  ⚠️ تنبيه 3: المعاملات القديمة بدون UUID                                 │
├──────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│  المعاملات المرتبطة بفواتير قديمة (قبل هذا التحديث) لا تحمل UUID.         │
│  هذه المعاملات لن تُزامَل عبر الحزمة المدمجة الجديدة.                     │
│                                                                          │
│  لو أردت مزامنة البيانات التاريخية بالكامل، قد تحتاج:                    │
│  - هجرة إضافية لتوليد UUID للمعاملات القديمة المرتبطة بفواتير            │
│  - أو الاكتفاء بالمعاملات الجديدة (من بعد التحديث)                       │
│                                                                          │
│  شباك الأمان المحاسبي (_recalculateCustomerBalanceInsideTxn) يضمن        │
│  أن الرصيد النهائي صحيح حتى لو نقصت بعض المعاملات التاريخية.             │
│                                                                          │
└──────────────────────────────────────────────────────────────────────────┘

┌──────────────────────────────────────────────────────────────────────────┐
│  ⚠️ تنبيه 4: رقم الجهاز يجب أن يكون فريداً                              │
├──────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│  نظام الترقيم يعتمد على أن رقم الجهاز (InvoiceSettingsService)            │
│  فريد لكل جهاز.                                                          │
│                                                                          │
│  لو تكرر رقم الجهاز على جهازين:                                          │
│  - كلاهما ينتج نفس نطاق الأرقام (مثل 1202681)                            │
│  - لكن القيد الفريد المركّب يمنع التعارض محلياً على نفس الجهاز            │
│  - التعارض فقط في رقم العرض (نفس الرقم على جهازين)                        │
│                                                                          │
│  الحل: التأكد من تعيين رقم جهاز مختلف لكل جهاز في الإعدادات.             │
│                                                                          │
└──────────────────────────────────────────────────────────────────────────┘
